#!/usr/bin/env bash
# 【第 4 段】装 Nginx + 写反向代理配置
set -e
echo "==> 安装 nginx 与 certbot（约 1-2 分钟，请等它跑完）"
apt-get update -qq
apt-get install -y nginx certbot python3-certbot-nginx

echo "==> 写反向代理配置"
cat > /etc/nginx/conf.d/rice-relay.conf <<'NGX'
server {
    listen 80;
    server_name rice-srt.cn;
    client_max_body_size 30m;
    client_body_timeout 120s;

    location /api/ {
        proxy_pass http://127.0.0.1:9000;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_read_timeout 120s;
        proxy_request_buffering off;
    }
}
NGX

nginx -t && systemctl reload nginx
echo "==> 第 4 段完成：Nginx 已把 http://rice-srt.cn/api/ 转给本机 9000"
curl -s http://127.0.0.1/api/ping; echo
